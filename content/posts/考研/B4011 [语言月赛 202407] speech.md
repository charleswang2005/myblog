#include <bits/stdc++.h>
using namespace std;
int main()
    {
    int n,m;
    cin>>n>>m;
    int a[1009]={0};
    int b[1009]={0};
    int c[1009]={0};
    for(int i=1;i<=m;i++)
        {
        cin>>a[i];
        }
    for(int i=1;i<=n;i++)
        {
        cin>>b[i];
        }
    for(int i=1;i<=n;i++)
        {
        c[b[i]]++;
        }
    for(int i=1;i<=m;i++)
        {
        c[i]*=a[i];
        }

        int max=0,ans=0;
        for(int i=1;i<=m;i++)
        {
        	if(c[i]>max)
        	{
        		max=c[i];
        		ans=i;
			}
		}
		cout<<ans<<endl;
    return 0;
    }
